## Usage

```tsx
import { List, ListItem } from "@ainativekit/ui";

<List
  items={contacts}
  renderItem={(contact) => (
    <ListItem
      key={contact.id}
      title={contact.name}
      subtitle={contact.email}
      media={contact.avatar}
    />
  )}
/>
```
