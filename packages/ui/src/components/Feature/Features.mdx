## Usage

```tsx
import { Features } from "@ainativekit/ui";
import { StarFilled, Clock } from "@openai/apps-sdk-ui/components/Icon";

// Simple strings
<Features items={["Neapolitan", "Wood-fired", "$$$"]} />

// With icons
<Features items={[
  { icon: <StarFilled />, label: "4.8" },
  { icon: <Clock />, label: "Open now" }
]} />
```

## Type Definition

```ts
type FeatureItem = string | { icon?: React.ReactNode; label: string };

interface FeaturesProps {
  items: FeatureItem[];
  separator?: string;  // default: "•"
  className?: string;
  style?: React.CSSProperties;
}
```

## Notes

- Used internally by Card, ListItem, and Map components
- Icons are passed as React elements (not icon names)
- Flexible separator supports any character or string
- Inherits text color from parent context
